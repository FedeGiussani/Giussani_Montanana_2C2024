176static void gesture_task(void *pvParameter)
177{
178 while (true)
179 {
180 LedToggle(LED_2);
181 ulTaskNotifyTake(pdTRUE, portMAX_DELAY); /* La tarea espera en este punto hasta recibir una notificación */
182 if (APDS9960_isGestureAvailable())
183 {
184 switch (APDS9960_readGesture())
185 {
186 case DIR_UP:
187 printf("up\r\n");
188 Duty_cycle = Duty_cycle + percent;
189 PWMSetDutyCycle(PWM_0, Duty_cycle);
190 break;
191 case DIR_DOWN:
192 printf("down\r\n");
193 Duty_cycle = Duty_cycle - percent;
194 PWMSetDutyCycle(PWM_0, Duty_cycle);
195 break;
196 case DIR_LEFT:;
197 printf("left\r\n");
198 PWMOff(PWM_0);
199 break;
200 case DIR_RIGHT:
201 printf("right\r\n");
202 PWMOn(PWM_0);
203 break;
204 case DIR_NEAR:
205 printf("near\r\n");
206 break;
207 case DIR_FAR:
208 printf("far\r\n");
209 break;
210 default:
211 printf("NONE\r\n");
212 }
213 }
214 }

233void app_main(void) {
234 // Inicialización de LEDs y GPIO
235 LedsInit();
236 GPIOInit(GPIO_1, GPIO_INPUT); // Configuración de GPIO_1 como entrada.
237 GPIOInit(GPIO_9, GPIO_OUTPUT); // Configuración de GPIO_9 como salida.
238
239 // Inicialización del sensor ultrasónico
240 HcSr04Init(GPIO_3, GPIO_2);
241
242 //Configuración del temporizador para la medición de distancia
243 timer_config_t timer_medicion = {
244 .timer = TIMER_A, // Temporizador seleccionado.
245 .period = REFRESCO_MEDICION, // Período de refresco de medición.
246 .func_p = FuncTimerMedir, // Función callback para el temporizador.
247 .param_p = NULL // Parámetro opcional para la función callback.
248 };
249 TimerInit(&timer_medicion);
250
251 // Configuración de interrupción para GPIO
252 GPIOActivInt(GPIO_1, pint_intr_callback, 0, NULL);
253 printf("Init APDS9960 test.\r\n");
254
255 // Inicialización de I2C y sensor APDS9960
256 I2C_initialize(100000); // Inicialización de I2C a 100 kHz.
257 if (!APDS9960_initialize())
258 {
259 printf("APDS9960 initialize failed.\r\n"); // Error de inicialización del sensor APDS9960.
260 }
261 APDS9960_enableGestureSensor(true); // Habilita el sensor de gestos APDS9960.
262
263 // Configuración de PWM en GPIO_9
264 PWMInit(PWM_0, GPIO_9, frec); // Inicialización del PWM con frecuencia especificada en `frec`.
265 printf("Init PWM.\r\n");
266
267 // Creación de tareas en FreeRTOS
268 xTaskCreate(&gesture_task, "GESTURE LOOP", 4096, NULL, 5, &gesture_process_event_handle); // Tarea para el procesamiento de gestos.
269 xTaskCreate(&OperarConDistancia, "medir", 4096, NULL, 5, &Medir_task_handle); // Tarea para la medición de distancia.
270
271 // Inicio del conteo de temporizadores
272 TimerStart(timer_medicion.timer); // Inicia el temporizador para la medición.
273}
